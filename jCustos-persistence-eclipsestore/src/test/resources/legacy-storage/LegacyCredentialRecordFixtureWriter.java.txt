import eu.jsentinel.jcustos.credential.store.CredentialRecord;
import eu.jsentinel.jcustos.credential.store.CredentialStatus;
import org.eclipse.store.storage.embedded.types.EmbeddedStorage;
import org.eclipse.store.storage.embedded.types.EmbeddedStorageManager;

import java.nio.file.Path;
import java.time.Instant;
import java.util.ArrayList;

/**
 * Writes the storage directory that {@code LegacyCredentialRecordCompatibilityTest} opens: an
 * application root - a list - holding credential records in the six-component shape of the
 * releases before 00.84.00, as an application that keeps its credential rows in its own
 * Eclipse-Store root would have stored them. Run against eu.jsentinel:jCustos-core:00.83.00.
 * Synthetic data only. Usage: {@code LegacyCredentialRecordFixtureWriter <empty directory>}.
 */
public final class LegacyCredentialRecordFixtureWriter {

  static final Instant T0 = Instant.parse("2026-03-01T08:00:00Z");

  private LegacyCredentialRecordFixtureWriter() {
  }

  public static void main(String[] args) {
    ArrayList<CredentialRecord> rows = new ArrayList<>();
    rows.add(new CredentialRecord("compromised", "$pwh$v=1$fixture-compromised",
        CredentialStatus.COMPROMISED, 3L, T0, T0.plusSeconds(60)));
    rows.add(new CredentialRecord("disabled", "$pwh$v=1$fixture-disabled",
        CredentialStatus.DISABLED, 2L, T0, T0.plusSeconds(30)));
    rows.add(new CredentialRecord("active", "$pwh$v=1$fixture-active",
        CredentialStatus.ACTIVE, 1L, T0, T0));
    EmbeddedStorageManager manager = EmbeddedStorage.start(rows, Path.of(args[0]));
    try {
      manager.storeRoot();
    } finally {
      manager.shutdown();
    }
  }
}
